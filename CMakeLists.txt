cmake_minimum_required(VERSION 3.16)
project(virtual_pet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core COMMAND core_tests)
add_executable(evolution_tests tests/evolution_tests.cpp)
add_test(NAME evolution COMMAND evolution_tests)
add_executable(multispecies_tests tests/multispecies_tests.cpp)
add_test(NAME multispecies COMMAND multispecies_tests)
if(WIN32)
 add_executable(virtual_pet WIN32 platforms/desktop/main.cpp)
 target_link_libraries(virtual_pet PRIVATE user32 gdi32 windowscodecs ole32 msimg32 shell32)
 add_executable(assets_tests tests/assets_tests.cpp)
 target_link_libraries(assets_tests PRIVATE user32 gdi32 windowscodecs ole32 msimg32)
 add_test(NAME assets COMMAND assets_tests "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
